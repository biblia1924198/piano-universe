cmake_minimum_required(VERSION 3.22.1)
project(PianoRoom C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/oboe)
add_library(pianoroom SHARED bridge.c oboe_adapter.cpp parallel_sampler.cpp)
target_compile_definitions(pianoroom PRIVATE PR_PARALLEL=1)
target_compile_options(pianoroom PRIVATE -O2 -fno-strict-aliasing)
target_link_options(pianoroom PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(pianoroom oboe m dl)
